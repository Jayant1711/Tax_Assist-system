\section{Tokenization guard: commas are grouping, not grammar}
\facts{\code{backend/nlp\_engine.py}{nlp\_engine.py:119} (grouping-safe split) ·
\code{backend/nlp\_engine.py}{nlp\_engine.py:112} (clause splitter).}

\begin{examplebox}
\textbf{Real use:} \emph{``claimed 1,25,000 under 80c''} must not split into three
clauses. The production regex splits only when neither neighbour of the comma is
a digit: \texttt{(?<!\textbackslash d)[,;](?!\textbackslash d)}. A naive split
destroys the amount into \texttt{1}, \texttt{25}, \texttt{000}.
\end{examplebox}

\section{Semantic attention (custom): distance-decayed category binding}
\facts{\code{backend/ai\_models.py}{ai\_models.py:142}
(\texttt{SemanticAttention}). \textbf{Mechanism:}
$s(k) = 100 \cdot e^{-d_k^2/2\sigma^2}$ with $\sigma = 40$ characters.
\textbf{Parameters:} zero learned; the decay scale is a fixed design constant.
\textbf{Evaluation:} covered by the extraction gate (F1 1.000) plus adversarial
unit tests for cross-category sentences.}

\begin{examplebox}
\textbf{Real use:} \emph{``invested 1.5 lakhs in PPF \ldots pay 2L rent''} ---
Section 80C scores at distance $d{=}6$ and wins the Rs 1.5 lakh; the rent
keyword scores at $d{=}34$ and wins the Rs 2 lakh. When nothing clears the
threshold, the amount falls to a flagged provisional default that can never
overwrite a booked value.
\end{examplebox}

\section{Intent classification: TF-IDF + logistic regression}
\facts{\code{nlp\_research/phase\_c\_intent\_classification.py}{phase\_c:73}
(feature build + the seeded split) · \code{nlp\_research/phase\_c\_intent\_classification.py}{phase\_c:135}
(training) · \code{nlp\_research/phase\_c\_intent\_classification.py}{phase\_c:300}
(runner). \textbf{Data:} 2,226 records $\rightarrow$ 1,780 train / 446 held-out
(stratified, seed 472; the TF-IDF vectorizer is fit on the TRAIN split only
--- the audit wave closed a full-corpus fit that leaked test vocabulary into
training features). 8 intents, TF-IDF unigrams+bigrams (min\_df 2, sublinear).
\textbf{Parameters:} 1,333 features $\times$ 8 classes $=$ \textbf{10,672
weights} (incl. intercepts; measured). \textbf{Training:} L-BFGS logistic
regression, \textbf{measured 9.7\,s wall} in the container, feature build
included. \textbf{Evaluation:} held-out \textbf{96.19\%} (train 98.48\%;
per-class report persisted in the artifact); the BiLSTM on the same split
measures \textbf{98.21\%}. The pre-audit ``97.9\% test accuracy'' was the
FULL-corpus score and is retracted (evaluation-integrity audit).}

\begin{examplebox}
\textbf{Real use:} \emph{``I am a software engineer earning 24 lakhs''} classifies
as the \texttt{salary} intent at very high margin, which routes the conversation
into the income branch. The classifier is research-layer: production routing
uses the same vocabulary through the deterministic path.
\end{examplebox}

\section{Sequence labeling with BIO tags (corpus encoding)}
\facts{\code{nlp\_research/phase\_d\_corpus.py}{phase\_d\_corpus.py:394}
(\texttt{entities\_to\_bio}) · \code{nlp\_research/phase\_d\_corpus.py}{phase\_d\_corpus.py:132}
(corpus generation) · \code{nlp\_research/phase\_d\_corpus.py}{phase\_d\_corpus.py:422}
(dataset preparation). \textbf{Data:} widened template grammar with fourteen
multi-entity sentence families; 500 unique annotated samples, deduplicated on
append. \textbf{Labels:} 11
BIO tags ($O$ plus B-/I- for each of the 5 entity types --- the former
DEDUCTION\_TYPE class merged into TAX\_SECTION; the old scheme double-labeled
every section token).}

\begin{examplebox}
\textbf{Real use:} \emph{``i paid 25k as rent on 1 april 2024 as of 30 june 2024''}
encodes as \texttt{O O B-MONEY O B-INCOME\_TYPE O B-DATE I-DATE I-DATE O O
B-DATE I-DATE I-DATE} --- four entities (``rent'' is an income word the shared
extractor labels too), three of them multi-token, aligned by character
offsets from \texttt{re.finditer} (the pre-fix window arithmetic produced
all-O and zero I- tags).
\end{examplebox}

\section{BiLSTM sequence labeling (NER model)}
\facts{\code{nlp\_research/phase\_d\_ner.py}{phase\_d\_ner.py:55} (model) ·
\code{nlp\_research/phase\_d\_ner.py}{phase\_d\_ner.py:91} (training loop) ·
\code{nlp\_research/phase\_d\_ner.py}{phase\_d\_ner.py:196} (evaluator).
\textbf{Data:} the 500-sample widened corpus above (14 multi-entity families
with explicit slot$\rightarrow$pool sequences; dedup-on-append; 64\% of the
sentences carry 2+ entities). \textbf{Parameters:}
\textbf{287,627} = embedding $161 \times 128$ (20,608, built from the TRAIN
rows only) + BiLSTM $2 \times
132{,}096$ (264,192) + linear $256 \times 11$ (2,827). \textbf{Training:}
Adam, lr $5{\times}10^{-4}$, gradient clipping at 1.0, 20 epochs ---
measured in the CPU container. \textbf{Evaluation:}
\textbf{99.14\% token accuracy, 0.994 macro F1} with the corrected evaluator
(the previously reported 0.143 was a softmax confidence of a NaN-diverged model,
with per-class numbers synthesized from entity counts). The label schema
shrank from six classes to five: the old scheme matched the same section
tokens as both TAX\_SECTION and DEDUCTION\_TYPE, so one span carried two gold
labels and the encoder silently kept the later one --- TAX\_SECTION gold had
collapsed to two tokens in the whole test split.}

\begin{examplebox}
\textbf{Real use:} research-track only --- production extraction stays
deterministic. The model exists so the low-data NER regime is measured
honestly, and the corrected pipeline is pinned by a torch-free test suite.
\end{examplebox}

\section{Training hygiene: gradient clipping}
\facts{\code{nlp\_research/phase\_d\_ner.py}{phase\_d\_ner.py:152}
(\texttt{clip\_grad\_norm\_} at 1.0). \textbf{Why:} the original NER training
diverged --- loss NaN at epoch 7, because the label encoding ignored every token
(all-O) so the cross-entropy reduced to 0/0. Clipping plus the corrected labels
removed the divergence entirely.}

\section{Word embeddings, built from scratch: GloVe}
\facts{\code{nlp\_research/embedding\_analysis.py}{embedding\_analysis.py:222}
(\texttt{train\_glove}). \textbf{Mechanism:} global co-occurrence factorization
with $f(x) = \min(1, (x/x_{max})^{\alpha})$ weighting, weighted least squares,
Adagrad. \textbf{Parameters:} 100-dimensional two-sided embeddings. \textbf{Why
from scratch:} no gensim wheel for the Python build --- the algorithms were
implemented in NumPy.}

capture 78\% of SGNS variance; t-SNE exposes fine clusters (80c / 80d / 24b).
\textbf{Documented finding:} template coverage anchors rare Hinglish tokens into
the right neighborhood.}

\section{Controlled falsification experiment design (three data scales)}
\facts{\code{backend/tools/forensics/}{tools/forensics} (the measurement
harness) · registry entries RUN-002 / RUN-003 / RUN-003b / RUN-004.
\textbf{Design:} identical architecture, monotonically more data (46,816
$\rightarrow$ 76,960 $\rightarrow$ 377,753 rows). \textbf{Measured outcomes:}
learned savings MAPE 92--108\% $\rightarrow$ 795\% $\rightarrow$
\textbf{1,523,679\%}; opportunity precision/recall 0 at every scale. \textbf{The
mechanism:} the label is a closed-form function containing a discontinuous
regime-flip $min()$ that additive rule bases smooth across. \textbf{Policy:}
learned regression on closed-form targets is permanently inadmissible.}
